Ability: BombardPlanet
	Name: #ABL_BOMBARD_PLANET
	Description: #ABL_BOMBARD_PLANET_DESC
	Icon: ActionBarIcons::2
	Hotkey: V

	Range: 80
	Target: targ = Object

	//Same four body types Conquer can target.
	Either(Either(Either(TargetFilterType(targ, Planet), TargetFilterType(targ, Asteroid)), TargetFilterType(targ, Star)), TargetFilterType(targ, Orbital))
	//Only enemy-owned bodies that still have a building standing --
	//there's nothing to bombard on a neutral or already-cleared body.
	TargetFilterOtherEmpire(targ)
	TargetFilterHasBuildings(targ)

	//No TargetFilterNotInCombat: bombarding is meant to happen during a
	//fight, unlike Conquer which requires the fighting to be over first.

	IsToggleTarget(targ, Channels = True)

	//Destroys one building every 3 seconds for as long as the ship stays
	//on target. Clear = False so it keeps re-firing instead of releasing
	//the target after the first hit; once the last building is gone,
	//TargetFilterHasBuildings makes the target invalid and the ability
	//lets go on its own. Conquer can then be used on the cleared body.
	AfterChannel(targ, 3, DestroyABuilding(), Clear = False)
